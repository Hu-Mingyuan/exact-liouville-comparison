  M.scalarDerivative (ExactLiouvilleComparison.ThirdPath.exhaustion P) x v =
    1 / (2 * √(P.R x)) * M.scalarDerivative P.R x v

    ∀ (v : M.Tangent x),
      ‖v‖ ≤ 1 → |M.scalarDerivative (ExactLiouvilleComparison.ThirdPath.exhaustion P) x v| ≤ 1 / (2 * √P.c)

  {Z : ↑FsFltz.Geometry.Parameter → (x : M.Point) → M.Tangent x},
  ExactLiouvilleComparison.MoserRegularity.NativeJoint Z →
    ∃ D,
      0 ≤ D ∧
        ∀ (s : ↑FsFltz.Geometry.Parameter) (x : M.Point),
          P.R x ≤ P.rstar → |M.scalarDerivative (ExactLiouvilleComparison.ThirdPath.exhaustion P) x (Z s x)| ≤ D
